# File 与 Base64 转换工具实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 新增一个离线、易用、支持任意文件与纯 Base64 双向转换的 `html/file2base64.html`。

**Architecture:** 页面保持为无依赖的单个 HTML 文件，并将可测试的转换函数暴露为 `window.FileBase64Core`。界面控制器只负责 DOM、文件读取、取消、复制和下载；二进制/Base64 转换、校验及文件名清理保持为纯函数或小型异步函数。

**Tech Stack:** HTML5、CSS3、原生 JavaScript、FileReader、Clipboard API、Blob/Object URL、Node.js `vm` 自动化测试。

- [ ] **Step 3：实现最小转换核心**

在 `html/file2base64.html` 中创建 `id="core-logic"` 的脚本，定义并导出以下完整 API：

```javascript
(function () {
  'use strict';
  const ENCODE_CHUNK_SIZE = 3 * 8192;
  const DECODE_CHUNK_SIZE = 4 * 8192;

  function cancelledError() {
    const error = new Error('转换已取消');
    error.code = 'CONVERSION_CANCELLED';
    return error;
  }

  function nextFrame() {
    return new Promise(resolve => setTimeout(resolve, 0));
  }

  async function bytesToBase64(bytes, onProgress, job) {
    if (job && job.cancelled) throw cancelledError();
    if (!bytes.length) {
      if (onProgress) onProgress(1);
      return '';
    }
    const parts = [];
    for (let offset = 0; offset < bytes.length; offset += ENCODE_CHUNK_SIZE) {
      if (job && job.cancelled) throw cancelledError();
      const chunk = bytes.subarray(offset, Math.min(offset + ENCODE_CHUNK_SIZE, bytes.length));
      let binary = '';
      for (let index = 0; index < chunk.length; index += 1) {
        binary += String.fromCharCode(chunk[index]);
      }
      parts.push(btoa(binary));
      if (onProgress) onProgress(Math.min(1, (offset + chunk.length) / bytes.length));
      if (parts.length % 16 === 0) await nextFrame();
    }
    return parts.join('');
  }

  function normalizeBase64(value) {
    return String(value || '').replace(/\s+/g, '');
  }

  function validateBase64(value) {
    const normalized = normalizeBase64(value);
    if (!normalized) return { valid: false, normalized, message: '请先粘贴 Base64 内容。' };
    if (normalized.length % 4 !== 0) return { valid: false, normalized, message: 'Base64 长度必须是 4 的倍数。' };
    const pattern = /^(?:[A-Za-z0-9+/]{4})*(?:[A-Za-z0-9+/]{2}==|[A-Za-z0-9+/]{3}=)?$/;
    if (!pattern.test(normalized)) return { valid: false, normalized, message: 'Base64 包含非法字符或填充。' };
    return { valid: true, normalized, message: '' };
  }

  async function base64ToBytes(value, onProgress, job) {
    const check = validateBase64(value);
    if (!check.valid) throw new Error(check.message);
    const chunks = [];
    let total = 0;
    for (let offset = 0; offset < check.normalized.length; offset += DECODE_CHUNK_SIZE) {
      if (job && job.cancelled) throw cancelledError();
      const binary = atob(check.normalized.slice(offset, offset + DECODE_CHUNK_SIZE));
      const chunk = new Uint8Array(binary.length);
      for (let index = 0; index < binary.length; index += 1) chunk[index] = binary.charCodeAt(index);
      chunks.push(chunk);
      total += chunk.length;
      if (onProgress) onProgress(Math.min(1, (offset + DECODE_CHUNK_SIZE) / check.normalized.length));
      if (chunks.length % 16 === 0) await nextFrame();
    }
    const result = new Uint8Array(total);
    let position = 0;
    chunks.forEach(chunk => { result.set(chunk, position); position += chunk.length; });
    return result;
  }

  function safeDownloadName(value) {
    const name = String(value || '').trim().replace(/[\\/:*?"<>|\x00-\x1F]/g, '_');
    if (!name || /^\.+$/.test(name)) return 'decoded.xlsx';
    return name;
  }

  function formatBytes(size) {
    if (!size) return '0 B';
    const units = ['B', 'KB', 'MB', 'GB'];
    const unit = Math.min(Math.floor(Math.log(size) / Math.log(1024)), units.length - 1);
    return `${Number((size / (1024 ** unit)).toFixed(unit ? 2 : 0))} ${units[unit]}`;
  }

  window.FileBase64Core = { bytesToBase64, base64ToBytes, normalizeBase64, validateBase64, safeDownloadName, formatBytes };
}());
```

- [ ] **Step 4：实现浏览器交互控制器**

控制器必须完成以下明确行为：

```javascript
const byId = id => document.getElementById(id);
let currentJob = null;

function beginJob() {
  cancelCurrentJob();
  currentJob = { cancelled: false, reader: null };
  return currentJob;
}

function cancelCurrentJob() {
  if (!currentJob) return;
  currentJob.cancelled = true;
  if (currentJob.reader && currentJob.reader.readyState === FileReader.LOADING) currentJob.reader.abort();
}

function showStatus(type, message) {
  const status = byId('status');
  status.className = `status status-${type}`;
  status.textContent = message;
  status.hidden = false;
}

function switchMode(mode) {
  const encode = mode === 'encode';
  byId('mode-encode').setAttribute('aria-selected', String(encode));
  byId('mode-decode').setAttribute('aria-selected', String(!encode));
  byId('panel-encode').hidden = !encode;
  byId('panel-decode').hidden = encode;
  byId('status').hidden = true;
}

function readFile(file, job, onProgress) {
  return new Promise((resolve, reject) => {
    const reader = new FileReader();
    job.reader = reader;
    reader.onprogress = event => {
      if (event.lengthComputable) onProgress(event.loaded / event.total * 0.35);
    };
    reader.onload = () => resolve(reader.result);
    reader.onerror = () => reject(new Error('无法读取所选文件。'));
    reader.onabort = () => {
      const error = new Error('转换已取消');
      error.code = 'CONVERSION_CANCELLED';
      reject(error);
    };
    reader.readAsArrayBuffer(file);
  });
}

async function handleFile(file) {
  if (!file) return;
  const job = beginJob();
  byId('base64-output').value = '';
  byId('copy-button').disabled = true;
  byId('download-text-button').disabled = true;
  byId('progress').hidden = false;
  byId('cancel-button').hidden = false;
  if (file.size > 50 * 1024 * 1024) showStatus('warning', '文件较大，转换期间浏览器会占用较多内存。');
  try {
    const buffer = await readFile(file, job, value => { byId('progress').value = value; });
    const base64 = await window.FileBase64Core.bytesToBase64(
      new Uint8Array(buffer),
      value => { byId('progress').value = 0.35 + value * 0.65; },
      job
    );
    if (job.cancelled) return;
    byId('base64-output').value = base64;
    byId('copy-button').disabled = false;
    byId('download-text-button').disabled = false;
    showStatus('success', `转换完成：${file.name}`);
  } catch (error) {
    byId('base64-output').value = '';
    showStatus(error.code === 'CONVERSION_CANCELLED' ? 'info' : 'error', error.message);
  } finally {
    if (currentJob === job) currentJob = null;
    byId('progress').hidden = true;
    byId('cancel-button').hidden = true;
  }
}

function downloadBlob(blob, filename) {
  const url = URL.createObjectURL(blob);
  const link = document.createElement('a');
  link.href = url;
  link.download = filename;
  document.body.appendChild(link);
  link.click();
  link.remove();
  setTimeout(() => URL.revokeObjectURL(url), 0);
}

async function decodeAndDownload() {
  const check = window.FileBase64Core.validateBase64(byId('decode-input').value);
  if (!check.valid) { showStatus('error', check.message); return; }
  const job = beginJob();
  try {
    const bytes = await window.FileBase64Core.base64ToBytes(check.normalized, null, job);
    downloadBlob(new Blob([bytes], { type: 'application/octet-stream' }), window.FileBase64Core.safeDownloadName(byId('download-name').value));
    showStatus('success', '文件已生成并开始下载。');
  } catch (error) {
    showStatus(error.code === 'CONVERSION_CANCELLED' ? 'info' : 'error', error.message);
  } finally {
    if (currentJob === job) currentJob = null;
  }
}

async function copyResult() {
  const output = byId('base64-output');
  try {
    if (!navigator.clipboard || !navigator.clipboard.writeText) throw new Error('fallback');
    await navigator.clipboard.writeText(output.value);
  } catch (error) {
    output.focus();
    output.select();
    if (!document.execCommand('copy')) { showStatus('error', '复制失败，请手动选择并复制。'); return; }
  }
  showStatus('success', '已复制到剪贴板。');
}
```

文件输入的 `change` 和拖放区的 `drop` 都调用 `handleFile(file)`。拖放区处理 `dragenter`、`dragover`、`dragleave`、`drop` 并阻止浏览器打开文件。转换开始时清除旧结果；成功后启用复制和文本下载；取消或失败后不得保留半成品结果。Base64 输入变化时清除旧成功状态。
